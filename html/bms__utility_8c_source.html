67 uint16_t bms_calcPec15(uint8_t *data, uint8_t len)
68 {
69  uint16_t remainder, addr;
70  remainder = 16; /* initialize the PEC */
71  for (uint8_t i = 0; i<len; i++) /* loops for each byte in data array */
72  {
73  addr = (((remainder>>7)^data[i])&0xff);/* calculate PEC table address */
74  remainder = ((remainder<<8)^Crc15Table[addr]);
75  }
76  return(remainder*2);/* The CRC15 has a 0 in the LSB so the remainder must be multiplied by 2 */
77 }

80 uint16_t bms_calcPec10(uint8_t *pDataBuf, int nLength, uint8_t *commandCounter)
81 {
82  uint16_t nRemainder = 16u; /* PEC_SEED */
83  /* x10 + x7 + x3 + x2 + x + 1 <- the CRC10 polynomial 100 1000 1111 */
84  uint16_t nPolynomial = 0x8Fu;
85  uint8_t nByteIndex, nBitIndex;
86  uint16_t nTableAddr;
87 
88  for (nByteIndex = 0u; nByteIndex < nLength; ++nByteIndex)
89  {
90  /* calculate PEC table address */
91  nTableAddr = (uint16_t)(((uint16_t)(nRemainder >> 2) ^ (uint8_t)pDataBuf[nByteIndex]) & (uint8_t)0xff);
92  nRemainder = (uint16_t)(((uint16_t)(nRemainder << 8)) ^ crc10Table[nTableAddr]);
93  }
94  /* If array is from received buffer add command counter to crc calculation */
95  if (commandCounter != NULL)
96  {
97  nRemainder ^= (uint16_t)(*commandCounter << 4u);
98  }
99  /* Perform modulo-2 division, a bit at a time */
100  for (nBitIndex = 6u; nBitIndex > 0u; --nBitIndex)
101  {
102  /* Try to divide the current data bit */
103  if ((nRemainder & 0x200u) > 0u)
104  {
105  nRemainder = (uint16_t)((nRemainder << 1u));
106  nRemainder = (uint16_t)(nRemainder ^ nPolynomial);
107  }
108  else
109  {
110  nRemainder = (uint16_t)((nRemainder << 1u));
111  }
112  }
113  return ((uint16_t)(nRemainder & 0x3FFu));
114 }

136 {
137  uint8_t txBuff_data[TOTAL_IC][DATAPKT_LEN]; // 6 Data + 2 DPEC per IC
138

140  {

151  }

158 void bms_spiReceiveData(uint8_t rxData[TOTAL_IC][DATA_LEN], uint16_t rxPec[TOTAL_IC], uint8_t rxCc[TOTAL_IC])
159 {
160  uint8_t rawRxData[TOTAL_IC][DATAPKT_LEN];
161 
162  HAL_SPI_Receive(hspi, (uint8_t *)rawRxData, DATAPKT_LEN * TOTAL_IC, HAL_MAX_DELAY);
163 
164  for (int ic = 0; ic < TOTAL_IC; ic++) /* executes for each ic in the daisy chain and packs the data */
165  {
166  // Store recieved data bytes to rxData
167  memcpy(rxData[ic], rawRxData[ic], DATA_LEN);
168 
169  // Get command counter value and store to the array
170  rxCc[ic] = rawRxData[ic][DATA_LEN] >> 2; // Get the 7th byte and shift right by 2 bits
171 
172  // Get received pec value from ic
173  // Mask the first 3 bits from 1st byte and combine with 2nd byte
174  rxPec[ic] = (uint16_t)(((rawRxData[ic][DATA_LEN] & 0x03) << 8) | rawRxData[ic][DATA_LEN + 1]);
175  }
176 }
