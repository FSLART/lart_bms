384  float sum = 0;
385 
386  for (int c = 0; c < TOTAL_CELL; c++) {

391  }
392  if (voltage < min) {
393  min = voltage;

395  }
396  ic_ad68[ic].v_avgCell_min = min;
397  ic_ad68[ic].v_avgCell_max = max;
398  ic_ad68[ic].v_avgCell_sum = sum;

400  ic_ad68[ic].v_avgCell_avg = sum / TOTAL_CELL;
401  ic_ad68[ic].v_avgCell_delta = max - min;

403  }
