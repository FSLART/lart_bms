14 static int cmp_desc(const void *a, const void *b) {
15  const float fa = *(const float*) a;
16  const float fb = *(const float*) b;
17  /* Place NaNs at the end (coldest) */
18  const int a_nan = !(fa == fa);
19  const int b_nan = !(fb == fb);
20  if (a_nan && b_nan)
21  return 0;
22  if (a_nan)
23  return 1;
24  if (b_nan)
25  return -1;
26  return (fb > fa) - (fb < fa); // descending
27 }

108 /* Discrete PID (positional form) with simple anti-windup via clamped integral.
109  u = Kp*e + Ki*Ts*sum(e) + Kd*(meas_prev - meas)/Ts (derivative on measurement)
110  Output is clamped to [out_min, out_max]. */
111 float PID_Update(pid_ctrl_t *pid, float measurement) {
112  if (!pid)
113  return 0.f;
114 
115  /* Sanitize measurement */
116  if (!(measurement == measurement) || !isfinite(measurement)) {
117  measurement = pid->setpoint; // neutralize crazy input
118  }
119 
120  const float e = pid->setpoint - measurement;
121 
122  /* Proportional */
123  const float P = pid->Kp * e;
124 
125  /* Derivative on measurement (noise-friendly) */
126  const float dmeas = (measurement - pid->prev_measurement);
127  const float D = (pid->Kd > 0.f && pid->Ts > 0.f) ? (-pid->Kd * dmeas / pid->Ts) : 0.f;
128 
129  /* Integral with basic anti-windup (clamp later using available headroom) */
130  pid->integrator += pid->Ki * pid->Ts * e;
131 
132  /* Pre-saturation sum */
133  float u = P + pid->integrator + D;
134 
135  /* Saturation */
136  if (u > pid->out_max)
137  u = pid->out_max;
138  if (u < pid->out_min)
139  u = pid->out_min;
140 
141  /* Anti-windup: back-calculate clamp by ensuring integrator stays within output range */
142  const float u_PI = P + pid->integrator; // exclude D (not stored)
143  if (u_PI > pid->out_max)
144  pid->integrator = pid->out_max - P;
145  else if (u_PI < pid->out_min)
146  pid->integrator = pid->out_min - P;
147 
148  pid->prev_error = e;
149  pid->prev_measurement = measurement;
150  return u;
151 }

155 float FanMgr_ComputeControlTemp(const float *temps, size_t num_temps, temp_ctrl_method_t method, size_t top_n) {
156  if (!temps || num_temps == 0)
157  return 0.f;
158 
159  /* Copy into a scratch buffer we can sort */
160  float *buf = (float*) malloc(num_temps * sizeof(float));
161  if (!buf)
162  return temps[0];
163  memcpy(buf, temps, num_temps * sizeof(float));
164 
165  float result = 0.f;
166 
167  switch (method) {
168  case TEMP_CTRL_MEAN_OF_TOP_N: {
169  if (top_n == 0 || top_n > num_temps)
170  top_n = num_temps;
171  qsort(buf, num_temps, sizeof(float), cmp_desc);
172  double sum = 0.0;
173  size_t count = 0;
174  for (size_t i = 0; i < top_n; ++i) {
175  if (!(buf[i] == buf[i]))
176  continue; // skip NaN
177  sum += buf[i];
178  count++;
179  }
180  result = (count > 0) ? (float) (sum / (double) count) : 0.f;
181  }
182  break;
183 
184  case TEMP_CTRL_MAX_OF_ALL: {
185  qsort(buf, num_temps, sizeof(float), cmp_desc);
186  result = buf[0];
187  if (!(result == result))
188  result = 0.f; // NaN guard
189  }
190  break;
191 
192  case TEMP_CTRL_MEDIAN_OF_ALL:
193  default: {
194  qsort(buf, num_temps, sizeof(float), cmp_asc);
195  /* Remove trailing NaNs */
196  size_t valid = num_temps;
197  while (valid > 0 && !(buf[valid - 1] == buf[valid - 1]))
198  valid--;
199  if (valid == 0)
200  result = 0.f;
201  else if ((valid & 1u) == 1u)
202  result = buf[valid / 2];
203  else
204  result = 0.5f * (buf[valid / 2 - 1] + buf[valid / 2]);
205  }
206  break;
207  }
208 
209  free(buf);
210  return result;
211 }
